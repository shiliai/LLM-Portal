快速开始

# 流式对话（任意 OpenAI 兼容客户端）
curl https://llm-portal.example.com/v1/chat/completions \
  -H "Authorization: Bearer sk-…" \
  -H "Content-Type: application/json" \
  -d '{"model":"deepseek-v4-flash-0731",
       "stream":true,
       "messages":[{"role":"user","content":"你好"}]}'
# Claude Code（Anthropic 协议自动转换）
export ANTHROPIC_BASE_URL=https://llm-portal.example.com
export ANTHROPIC_AUTH_TOKEN=sk-…
# 默认模型名 claude-opus-5 已映射到 deepseek

# pi CLI（已内置 private-llm provider）
pi --provider private-llm --model qwen3.6-35b-fp8

MCP 视觉工具（同一把 Key）

# MCP 端点：Streamable HTTP + Bearer
https://llm-portal.example.com/mcp
# 内建工具 analyze_image(image_url, question)——token 记在调用者 Key 上
# 本地图片两步式：先上传换 30 分钟临时 URL，再调用工具
curl -X POST https://llm-portal.example.com/mcp/upload \
  -H "Authorization: Bearer sk-…" -F file=@shot.png
# 用量自查
curl https://llm-portal.example.com/mcp/usage -H "Authorization: Bearer sk-…"
